```sql
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_document_chunks_embedding ON document_chunks USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);

    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID REFERENCES sessions(id) ON DELETE CASCADE,
    identifier VARCHAR(32) NOT NULL, -- 'agent_a', 'agent_b', 'judge'
    name VARCHAR(64) NOT NULL,
    role_title VARCHAR(128) NOT NULL,
    bias_direction VARCHAR(32) NOT NULL, -- 'FOR', 'AGAINST', 'NEUTRAL'

    avatar_color VARCHAR(16) NOT NULL DEFAULT '#6366F1',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE turns (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    unresolved_risks JSONB DEFAULT '[]',

    action_plan JSONB DEFAULT '[]',
```

```python
from typing import List, Dict, Any, Optional
from pydantic import BaseModel, Field

class Citation(BaseModel):
    source_title: str
    page_number: Optional[int] = None
    url: Optional[str] = None
    snippet: str

class FallacyAlert(BaseModel):
    fallacy_type: str  # "STRAWMAN", "AD_HOMINEM", "FALSE_DICHOTOMY", "SLIPPERY_SLOPE", "CIRCULAR"
    quote: str
    explanation: str

class MessageTurn(BaseModel):
    id: str
    speaker_id: str
    sender_name: str
    role_title: str
    round_number: int
    phase: str  # "FOUNDATION", "EVIDENCE", "CLASH", "INTERJECTION"
    content: str
    citations: List[Citation] = Field(default_factory=list)
    fallacies: List[FallacyAlert] = Field(default_factory=list)

class Persona(BaseModel):
    id: str
    identifier: str  # "agent_a", "agent_b", "judge"
    name: str
    role_title: str
    bias_direction: str  # "FOR", "AGAINST", "NEUTRAL"
    core_worldview: str
    system_prompt: str
    avatar_color: str = "#6366F1"

class HumanInterjection(BaseModel):
    user_name: str = "Moderator"
    content: str
    target_persona_id: Optional[str] = None

class DebateState(BaseModel):
    session_id: str
    topic: str
    mode: str = "DEBATE"
    document_context: Optional[str] = None
    personas: List[Persona] = Field(default_factory=list)
    turns: List[MessageTurn] = Field(default_factory=list)
    current_round: int = 1
    total_rounds: int = 5
    active_persona_idx: int = 0
    is_completed: bool = False
    pending_interjections: List[HumanInterjection] = Field(default_factory=list)
    verdict: Optional[Dict[str, Any]] = None
```

### 6.2. LangGraph Cyclic Engine (backend/app/engine/graph.py)
```python
from langgraph.graph import StateGraph, END
from app.engine.state import DebateState
from app.engine.nodes.synthesizer import synthesize_personas_node
from app.engine.nodes.debater import debater_turn_node
from app.engine.nodes.referee import referee_audit_node
from app.engine.nodes.interjection import apply_interjections_node
from app.engine.nodes.judge import judge_adjudication_node

def route_next_step(state: DebateState) -> str:
    # 2 turns per round (Agent A then Agent B)
    if state.is_completed or (state.current_round > state.total_rounds):
        return "judge"
    if len(state.pending_interjections) > 0:
        return "apply_interjections"
    return "debater"

def build_debate_graph():
    workflow = StateGraph(DebateState)

    workflow.add_node("synthesizer", synthesize_personas_node)
    workflow.add_node("debater", debater_turn_node)
    workflow.add_node("referee", referee_audit_node)
    workflow.add_node("apply_interjections", apply_interjections_node)
    workflow.add_node("judge", judge_adjudication_node)

    # Initial flow
    workflow.set_entry_point("synthesizer")
    workflow.add_edge("synthesizer", "debater")
    workflow.add_edge("debater", "referee")
    
    # Cyclic router
    workflow.add_conditional_edges(
        "referee",
        route_next_step,
        {
            "debater": "debater",
            "apply_interjections": "apply_interjections",
            "judge": "judge"
        }
    )
    workflow.add_edge("apply_interjections", "debater")
    workflow.add_edge("judge", END)

    return workflow.compile()
```

### 6.3. Dynamic Persona Synthesizer (backend/app/engine/nodes/synthesizer.py)
```python
import json
import uuid
from langchain_core.messages import SystemMessage, HumanMessage
from app.engine.state import DebateState, Persona

SYNTHESIZER_SYSTEM_PROMPT = """
You are an expert Debate Architect and Dialectic Systems Designer.
Analyze the provided topic and reference document context.
Identify the primary underlying tension (e.g. Innovation vs Compliance,
Security vs Velocity, High CapEx vs Capital Efficiency, Freedom vs Regulation).

Design two opposing world-class expert personas who would authentically clash
over this topic, plus an impartial senior decision-maker as the judge.

Output strictly valid JSON with this schema:
{
  "personas": [
    {
      "identifier": "agent_a",
      "name": "First Last",
      "role_title": "Specific Professional Title",
      "bias_direction": "FOR",
      "core_worldview": "Short summary of their core perspective",
      "avatar_color": "#4F46E5",
      "system_prompt": "Direct instructions for their debate persona, priorities, and debate style."
    },
    {
      "identifier": "agent_b",
      "name": "First Last",
      "role_title": "Specific Professional Counter-Title",
      "bias_direction": "AGAINST",
      "core_worldview": "Short summary of their opposing perspective",
      "avatar_color": "#DC2626",
      "system_prompt": "Direct instructions for their counter-persona, priorities, and debate style."
    }
  ]
}
"""

async def synthesize_personas_node(state: DebateState, llm) -> DebateState:
    context = f"Topic: {state.topic}\n\nDocument Reference:\n{state.document_context or 'None provided.'}"
    response = await llm.ainvoke([
        SystemMessage(content=SYNTHESIZER_SYSTEM_PROMPT),
        HumanMessage(content=context)
    ])
    
    parsed = json.loads(response.content)
    personas_list = []
    for p in parsed["personas"]:
        personas_list.append(Persona(
            id=str(uuid.uuid4()),
            identifier=p["identifier"],
            name=p["name"],
            role_title=p["role_title"],
            bias_direction=p["bias_direction"],
            core_worldview=p["core_worldview"],
            system_prompt=p["system_prompt"],
            avatar_color=p.get("avatar_color", "#4F46E5")
        ))
    state.personas = personas_list
    return state
```

### 6.4. Turn Execution with Phased Guidance (backend/app/engine/nodes/debater.py)
```python
import uuid
from langchain_core.messages import SystemMessage, HumanMessage
from app.engine.state import DebateState, MessageTurn
from app.engine.tools.rag_search import retrieve_semantic_context

def determine_phase_prompt(current_round: int, total_rounds: int) -> tuple[str, str]:
    progress = current_round / total_rounds
    if progress <= 0.33:
        return (
            "FOUNDATION",
            "PHASE: FOUNDATION. Clearly define your terms, boundary conditions, and primary logical premise. "
            "Do not hand-wave. Establish the core ethical, technical, or financial thesis."
        )
    elif progress <= 0.66:
        return (
            "EVIDENCE",
            "PHASE: EVIDENCE & CROSS-EXAMINATION. You MUST back your claims with concrete empirical precedent, "
            "metrics, or data points from the provided context. Pure abstract rhetoric will be penalized."
        )
    else:
        return (
            "CLASH",
            "PHASE: DIRECT CLASH & DECONSTRUCTION. Target the opponent's previous argument. Expose flaws in their "
            "assumptions, hidden costs, or logical leaps. Do not simply restate your earlier points."
        )

async def debater_turn_node(state: DebateState, llm, db_session) -> DebateState:
    active_persona = state.personas[state.active_persona_idx % 2]
    phase_name, phase_instruction = determine_phase_prompt(state.current_round, state.total_rounds)

    # 1. Retrieve RAG grounding chunks if document was uploaded
    rag_context = ""
    if state.document_context:
        # Search relevant chunks using last opponent message or topic as query
        query = state.turns[-1].content if state.turns else state.topic
        chunks = await retrieve_semantic_context(state.session_id, query, db_session, top_k=3)
        if chunks:
            rag_context = "\n### RELEVANT DOCUMENT CONTEXT ###\n" + "\n".join(
                f"[Source: {c['file_name']}, Page {c['page_number']}]: {c['content']}" for c in chunks
            )

    # 2. Build Sliding Conversation Window (last 4 turns)
    recent_history = "\n\n".join(
        f"[{t.role_title} - {t.sender_name}]:\n{t.content}"
        for t in state.turns[-4:]
    )

    full_system = f"{active_persona.system_prompt}\n\n{phase_instruction}\n{rag_context}"
    user_prompt = f"TOPIC: {state.topic}\n\nRECENT DIALECTIC HISTORY:\n{recent_history or 'Opening statement.'}\n\nYOUR RESPONSE:"

    response = await llm.ainvoke([
        SystemMessage(content=full_system),
        HumanMessage(content=user_prompt)
    ])

    new_turn = MessageTurn(
        id=str(uuid.uuid4()),
        speaker_id=active_persona.identifier,
        sender_name=active_persona.name,
        role_title=active_persona.role_title,
        round_number=state.current_round,
        phase=phase_name,
        content=response.content,
        citations=[],
        fallacies=[]
    )

    state.turns.append(new_turn)
    state.active_persona_idx += 1
    if state.active_persona_idx % 2 == 0:
        state.current_round += 1

    return state
```

### 6.5. RAG & Grounding Engine (backend/app/engine/tools/rag_search.py)
```python
from sqlalchemy import text
import openai

async def generate_embedding(text_content: str) -> list[float]:
    # Replace with OpenAI or Gemini embedding call
    response = await openai.AsyncOpenAI().embeddings.create(
        model="text-embedding-3-small",
        input=text_content
    )
    return response.data[0].embedding

async def retrieve_semantic_context(session_id: str, query: str, db_session, top_k: int = 3) -> list[dict]:
    query_vector = await generate_embedding(query)
    
    stmt = text("""
        SELECT file_name, page_number, content, 
               1 - (embedding <=> :vector::vector) AS similarity
        FROM document_chunks
        WHERE session_id = :session_id
        ORDER BY similarity DESC
        LIMIT :top_k
    """)
    
    result = await db_session.execute(stmt, {
        "vector": str(query_vector),
        "session_id": session_id,
        "top_k": top_k
    })
    
    rows = result.fetchall()
    return [{"file_name": r[0], "page_number": r[1], "content": r[2], "similarity": float(r[3])} for r in rows]
```

### 6.6. Real-Time Fallacy Referee (backend/app/engine/nodes/referee.py)
```python
import json
from langchain_core.messages import SystemMessage, HumanMessage
from app.engine.state import DebateState, FallacyAlert

REFEREE_PROMPT = """
You are an uncompromising Formal Logician and Debate Referee.
Examine the active speaker's latest response in light of the opponent's previous statement.
Detect any of the following informal fallacies:
- STRAWMAN: Misrepresenting or caricaturing the opponent's claim.
- AD_HOMINEM: Attacking persona traits or motives instead of substance.
- FALSE_DICHOTOMY: Presenting a false either/or binary.
- SLIPPERY_SLOPE: Claiming an extreme, unjustified chain of disasters.
- CIRCULAR_REASONING: Assuming the conclusion in the premise.

Output strictly valid JSON with this schema:
{
  "fallacies": [
    {
      "fallacy_type": "STRAWMAN | AD_HOMINEM | FALSE_DICHOTOMY | SLIPPERY_SLOPE | CIRCULAR_REASONING",
      "quote": "Exact verbatim quote from the active message",
      "explanation": "Concise 1-sentence breakdown of why this is logically flawed."
    }
  ]
}
If no fallacies are present, return { "fallacies": [] }.
"""

async def referee_audit_node(state: DebateState, fast_llm) -> DebateState:
    if len(state.turns) < 2:
        return state

    latest_turn = state.turns[-1]
    preceding_turn = state.turns[-2]

    context = f"OPPONENT SAID:\n{preceding_turn.content}\n\nACTIVE SPEAKER REPLIED:\n{latest_turn.content}"

    try:
        response = await fast_llm.ainvoke([
            SystemMessage(content=REFEREE_PROMPT),
            HumanMessage(content=context)
        ])
        data = json.loads(response.content)
        for f in data.get("fallacies", []):
            latest_turn.fallacies.append(FallacyAlert(
                fallacy_type=f["fallacy_type"],
                quote=f["quote"],
                explanation=f["explanation"]
            ))
    except Exception as e:
        print(f"Referee audit skipped: {e}")

    return state
```

```python
Output strictly valid JSON with this schema:

  "winner_identifier": "agent_a | agent_b | TIE",

  },
  "key_concessions": ["..."],
  "unresolved_risks": ["..."],
  "executive_recommendation": "...",
  "action_plan": ["Step 1", "Step 2", "Step 3"]
}
"""

async def judge_adjudication_node(state: DebateState, frontier_llm) -> DebateState:
    full_transcript = "\n\n".join(
        f"[{t.role_title} - {t.sender_name} (Round {t.round_number} - {t.phase})]:\n{t.content}"
        for t in state.turns
    )
    
    response = await frontier_llm.ainvoke([
        SystemMessage(content=JUDGE_PROMPT),
        HumanMessage(content=f"TOPIC: {state.topic}\n\nFULL TRANSCRIPT:\n{full_transcript}")
    ])
```

### 6.9. Real-Time Server-Sent Events API (backend/app/api/routes_stream.py)
```python
from fastapi import APIRouter
from fastapi.responses import StreamingResponse
import asyncio
import json

router = APIRouter()

@router.get("/api/sessions/{session_id}/stream")
async def stream_debate(session_id: str):
    async def event_generator():
        # Retrieve active state or load from DB
        state = await load_debate_state(session_id)
        
        # 1. Emit Initial Personas
        yield f"event: personas_ready\ndata: {json.dumps([p.dict() for p in state.personas])}\n\n"

        while not state.is_completed:
            active_persona = state.personas[state.active_persona_idx % 2]
            
            # 2. Turn Start Event
            yield f"event: turn_start\ndata: {json.dumps({'speaker_id': active_persona.identifier, 'name': active_persona.name, 'round': state.current_round})}\n\n"

            # 3. Stream Tokens Character-by-Character
            async for token in stream_agent_tokens(state, active_persona):
                yield f"event: token\ndata: {json.dumps({'token': token})}\n\n"
                await asyncio.sleep(0.015) # Paced delivery

            # 4. Fallacy & Citation Audit Emit
            turn = state.turns[-1]
            if turn.fallacies:
                for f in turn.fallacies:
                    yield f"event: fallacy_alert\ndata: {json.dumps(f.dict())}\n\n"

            yield f"event: turn_end\ndata: {json.dumps({'turn_id': turn.id})}\n\n"
            
            # Step graph forward
            await advance_graph_step(state)
            await asyncio.sleep(1.0) # Organic human-like pause

        # 5. Final Adjudication
        yield f"event: verdict_ready\ndata: {json.dumps(state.verdict)}\n\n"

    return StreamingResponse(event_generator(), media_type="text/event-stream")
```

### 7.1. Streaming EventSource Consumer (frontend/src/hooks/useDebateStream.ts)
```typescript
import { useState, useEffect, useRef } from 'react';
import { MessageTurn, Persona, Verdict, FallacyAlert } from '../lib/types';

export function useDebateStream(sessionId: string) {
  const [personas, setPersonas] = useState<Persona[]>([]);
  const [turns, setTurns] = useState<MessageTurn[]>([]);
  const [activeSpeakerId, setActiveSpeakerId] = useState<string | null>(null);
  const [currentStreamedText, setCurrentStreamedText] = useState<string>('');
  const [fallacies, setFallacies] = useState<FallacyAlert[]>([]);
  const [verdict, setVerdict] = useState<Verdict | null>(null);
  const [isFinished, setIsFinished] = useState<boolean>(false);

  useEffect(() => {
    if (!sessionId) return;

    const eventSource = new EventSource(`/api/sessions/${sessionId}/stream`);

    eventSource.addEventListener('personas_ready', (e) => {
      setPersonas(JSON.parse(e.data));
    });

    eventSource.addEventListener('turn_start', (e) => {
      const data = JSON.parse(e.data);
      setActiveSpeakerId(data.speaker_id);
      setCurrentStreamedText('');
    });

    eventSource.addEventListener('token', (e) => {
      const { token } = JSON.parse(e.data);
      setCurrentStreamedText((prev) => prev + token);
    });

    eventSource.addEventListener('fallacy_alert', (e) => {
      const fallacy: FallacyAlert = JSON.parse(e.data);
      setFallacies((prev) => [...prev, fallacy]);
    });

    eventSource.addEventListener('turn_end', (e) => {
      setCurrentStreamedText('');
      setActiveSpeakerId(null);
      // Fetch latest completed turns or append
    });

    eventSource.addEventListener('verdict_ready', (e) => {
      setVerdict(JSON.parse(e.data));
      setIsFinished(true);
      eventSource.close();
    });

    eventSource.onerror = (err) => {
      console.error('SSE Error:', err);
      eventSource.close();
    };

    return () => eventSource.close();
  }, [sessionId]);

  return { personas, turns, activeSpeakerId, currentStreamedText, fallacies, verdict, isFinished };
}
```
